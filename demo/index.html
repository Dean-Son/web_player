사용 방법

설치

npm install dean-video-player

기본 사용법

import { VideoPlayer } from 'dean-video-player';

const player = new VideoPlayer({
  container: '#video-container',
  src: 'https://example.com/video.mp4', // HTTPS 권장
  poster: 'https://example.com/poster.jpg',
  width: 800,        // 픽셀 단위
  height: 450,       // 픽셀 단위
  aspectRatio: '16:9', // 화면 비율
  autoplay: false,
  muted: false,
  events: {
    error: (error) => {
      console.error('플레이어 오류:', error.message);
      // 사용자에게 친화적인 오류 메시지 표시
    },
    loadstart: () => console.log('비디오 로드 시작'),
    canplay: () => console.log('재생 준비 완료')
  }
});

실시간 미리보기 (자동)

// 설정 없이 자동으로 실시간 프레임 캡처
const player = new VideoPlayer({
  container: '#video-container',
  src: 'video.mp4'
});

// 선택적으로 정적 이미지와 혼합 사용
const player = new VideoPlayer({
  container: '#video-container',
  src: 'video.mp4',
  previewImages: [
    { time: 0, url: 'custom_thumbnail.jpg' }, // 특정 시간만 커스텀 이미지
    // 나머지 시간은 자동으로 실시간 캡처
  ]
});

크기 조절 API

// 런타임에 크기 변경
player.setSize(800, 450);           // 픽셀 단위
player.setSize('100%', '50vh');     // CSS 단위
player.setSize();                   // 기본 크기로 복원
player.setAspectRatio('21:9');      // 화면 비율 변경

이벤트 리스너 및 에러 처리

// 생성 시 이벤트 설정
const player = new VideoPlayer({
  container: '#video-container',
  src: 'https://example.com/video.mp4',
  events: {
    play: () => console.log('재생 시작'),
    pause: () => console.log('일시정지'),
    timeupdate: (time) => console.log('현재 시간:', time),
    error: (error) => {
      // 에러 처리
      console.error('비디오 오류:', error);
      showUserFriendlyError('비디오를 재생할 수 없습니다.');
    },
    loadstart: () => console.log('로드 시작'),
    loadedmetadata: () => console.log('메타데이터 로드 완료'),
    canplay: () => console.log('재생 가능')
  }
});

// 안전한 비디오 소스 변경
try {
  player.setSrc('https://new-video-url.mp4');
} catch (error) {
  console.error('소스 변경 실패:', error.message);
}

접근성 및 키보드 네비게이션

// 플레이어는 자동으로 접근성 기능을 제공합니다:
// - ARIA 라벨 및 역할
// - 키보드 네비게이션 (Tab, Space, 화살표 키)
// - 스크린 리더 지원
// - 동적 상태 업데이트

// 추가 접근성 설정이 필요한 경우:
const container = document.getElementById('video-container');
container.setAttribute('aria-label', '메인 비디오 플레이어');

// 키보드 이벤트 확장 (선택사항)
container.addEventListener('keydown', (e) => {
  if (e.code === 'KeyR') {
    e.preventDefault();
    player.setCurrentTime(0); // R키로 처음으로
  }
});

키보드 단축키 및 접근성

Space - 재생/일시정지
← - 10초 뒤로
→ - 10초 앞으로
↑ - 볼륨 증가
↓ - 볼륨 감소
M - 음소거 토글
F - 전체화면 토글
Home - 처음으로 이동
End - 끝으로 이동
Tab - 컨트롤 요소 간 이동
스크린 리더 지원 (ARIA 라벨)
키보드 전용 네비게이션 지원
